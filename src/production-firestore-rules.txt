rules_version = '2';

service cloud.firestore {
  match /databases/{database}/documents {

    match /users/{userId} {
      allow read, write: if request.auth != null
                         && request.auth.uid == userId;
    match /completed/{animeId} {
  allow read, write: if request.auth != null
                     && request.auth.uid == userId;
}

      match /watchlist/{animeId} {
        allow read, write: if request.auth != null
                           && request.auth.uid == userId;
      }

      match /collections/{collectionId} {
        allow read, write: if request.auth != null
                           && request.auth.uid == userId;

        match /items/{animeId} {
          allow read, write: if request.auth != null
                             && request.auth.uid == userId;
        }
      }
    }
  }
}
